import numpy as np

# Define feature matrix X for the housing data
X = np.array([[1000, 2, 10], 
              [1500, 3, 5], 
              [1200, 2, 8], 
              [1800, 4, 2]])

# 1. Mean centering the matrix
X_centered = X - np.mean(X, axis=0)

# 2. Calculating the Covariance Matrix
covariance_matrix = np.cov(X, rowvar=False)

# 3. Print the result
print("Covariance Matrix:")
print(covariance_matrix)
